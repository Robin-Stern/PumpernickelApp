```kotlin
class FakeSecureKeyStore(
    initialLegacyKey: String? = null,
    var failOnRead: Boolean = false
) : SecureKeyStoreSurface {
    private val store = mutableMapOf<String, Credential>()
    var legacyKey: String? = initialLegacyKey
    val writeCalls = mutableListOf<Pair<ProviderId, Credential>>()
    var clearLegacyCallCount = 0

    suspend fun writeCredential(provider: ProviderId, credential: Credential) {
        store[provider.apiKeyAccount] = credential
        writeCalls += (provider to credential)
        if (provider == ProviderId.OpenAI && credential is Credential.ApiKey) {
            legacyKey = credential.value   // same physical slot
        }
    }
    suspend fun readCredential(provider: ProviderId): Credential? = store[provider.apiKeyAccount]
    suspend fun clearCredential(provider: ProviderId) {

        if (provider == ProviderId.OpenAI) legacyKey = null

        ProviderId.entries.firstOrNull { it.apiKeyAccount == key }

        if (failOnRead) throw RuntimeException("simulated keychain failure")
        return legacyKey
    }
    suspend fun clearLegacyApiKey() {
        clearLegacyCallCount++
        legacyKey = null
        store.remove(ProviderId.OpenAI.apiKeyAccount)
    }
}
```

```kotlin
package com.pumpernickel.data.api

import com.pumpernickel.domain.ai.AiError
import kotlinx.serialization.json.Json

/**
 * D-22-11 — pure-Kotlin Anthropic SSE parser. Stateful (carries currentEventType
 * across calls). Decoupled from Ktor for testability.
 *
 * Used by [AnthropicClient.chatCompletionStreaming] inside its channel-read loop.
 */
class AnthropicSseParser(
    private val json: Json = Json { ignoreUnknownKeys = true; encodeDefaults = true }
) {
    private var currentEventType: String? = null
    private val content = StringBuilder()
    var done: Boolean = false
        private set

    /**
     * Feed one SSE line. Returns the new accumulated `content` if it changed,
     * or null otherwise. Throws AiError on error-event frames.
     */
    fun feed(line: String): String? {
        if (line.isBlank()) { currentEventType = null; return null }
        when {
            line.startsWith("event:") -> { currentEventType = line.substring(6).trim(); return null }
            line.startsWith("data:") -> {
                val payload = line.substring(5).trim()
                when (currentEventType) {
                    "content_block_delta" -> {
                        val frame = tryDecode<AnthropicContentBlockDeltaFrame>(payload) ?: return null
                        if (frame.delta.type == "text_delta" && frame.delta.text.isNotEmpty()) {
                            content.append(frame.delta.text)
                            return content.toString()
                        }
                    }
                    "error" -> {
                        val frame = tryDecode<AnthropicErrorFrame>(payload)
                        val errType = frame?.error?.type ?: ""
                        val msg = frame?.error?.message ?: payload.take(200)
                        throw when {
                            errType.endsWith("authentication_error") -> AiError.AuthOrQuota(401)
                            errType.endsWith("overloaded_error") -> AiError.Provider(529)
                            else -> AiError.SchemaInvalid("Anthropic stream error: $msg")
                        }
                    }
                    "message_stop" -> { done = true; return null }
                    else -> { /* ignore message_start, content_block_start/stop, message_delta, ping */ }
                }
                return null
            }
            else -> return null
        }
    }

    fun result(): String = content.toString()

    private inline fun <reified T> tryDecode(payload: String): T? = try {
        json.decodeFromString<T>(payload)
    } catch (_: Throwable) { null }
}
```
